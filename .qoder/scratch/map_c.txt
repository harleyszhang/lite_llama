@@@FILE lite_llama/kernels/backend/deepgemm/linear.py

``fp8_gemm_nt`` repacks the weight into DeepGEMM's NT layout via

Usage:

@@@FILE lite_llama/kernels/backend/deepgemm/moe.py

each expert actually holds.

Usage:
    y = grouped_moe(hidden_states, w1, w2, topk_weights, topk_ids)

@@@FILE lite_llama/kernels/backend/deepgemm/quant.py

@@@FILE lite_llama/kernels/dispatcher/autotune/__init__.py
"""Kernel autotune: collect, persist, and look up optimal tile configurations.

:func:`get_best_config` is the call-site entry point; :class:`ConfigStore`

Usage:
    from lite_llama.kernels.dispatcher.autotune import get_best_config

@@@END
@@@FILE lite_llama/kernels/dispatcher/autotune/config_key.py
"""Autotune key structure — the stable contract behind a stored config.

:class:`TuneKey` normalises (op, m/n/k buckets, dtype, GPU name) into one
hashable key, so a config measured for one shape or machine is never
silently reused for another.

Usage:
    key = TuneKey.build("fused_moe", 16, 4096, 4096, "fp16", gpu_name)
"""
@@@END
@@@FILE lite_llama/kernels/dispatcher/autotune/config_store.py
"""Persistent JSON config store for autotune results.

:class:`ConfigStore` reads and writes one JSON document per cache dir,
merging measured configs under their :class:`TuneKey` — simple enough
to inspect by hand, stable enough to commit.

Usage:
    store = ConfigStore(cache_dir)
"""
@@@END

@@@FILE lite_llama/kernels/dispatcher/dispatch.py
"""Deterministic dispatch: filter, rank, cache — and be able to say why.

:func:`dispatch` builds a :class:`DispatchKey` from the call context,
filters the registry's specs to feasible ones, ranks them (native floor
unless a perf provider says better) and returns a :class:`Selected`
whose ``explain()`` states the reason.

Usage:
    selected = dispatch("rmsnorm"); print(selected.explain())
    kernel = selected.load()
"""
@@@END

Usage:
    from lite_llama.kernels.dispatcher import register; register(spec)

@@@END
@@@FILE lite_llama/kernels/dispatcher/spec.py
"""Declarative KernelSpec: one implementation's dispatch contract, torch-free.

A :class:`KernelSpec` names its op and backend target, states its shape /
layout requirements, and carries golden records plus perf tags; it is
pure data, so the registry can filter rows without importing torch.

Usage:
    spec = KernelSpec(...); register(spec)

@@@FILE lite_llama/kernels/ops/quantization/w8a8.py
