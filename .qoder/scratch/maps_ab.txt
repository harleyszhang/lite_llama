@@@FILE lite_llama/engine/async_engine.py
"""Asyncio front end for the continuous-batching engine.

:class:`AsyncLLMEngine` runs one pump thread that drives the synchronous
engine's ``step()`` loop and fans results out to per-request asyncio
streams, so many callers share a single batching loop.

Usage:
    engine = AsyncLLMEngine(sync_engine)
    async for chunk in await engine.generate(prompt): ...

@@@FILE lite_llama/engine/continuous_engine.py
"""Continuous batching: a step-driven engine where requests join and leave freely.

Each ``step()`` asks the :class:`~lite_llama.engine.scheduler.Scheduler` for
a plan, runs it through the executor, harvests sampled tokens and updates
request state — so chunked prefills and running decodes share one pass.

Usage:
    engine.add_request(prompt, params)
    finished = engine.step()
"""
@@@END
@@@FILE lite_llama/engine/data_parallel.py
"""Data parallelism: N whole-model replicas, one process each, routed between.

:class:`DataParallelEngine` is the parent half: it spawns one ``_dp_worker``
process per replica, routes every request through a
:class:`~lite_llama.engine.dp_load_balancer.LoadBalancer`, and merges the
replicas' results into one ordered output stream.

Usage:
    engine = DataParallelEngine(model, data_parallel_size=2)
    outs = engine.generate(prompts, sampling_params)
"""

@@@FILE lite_llama/engine/detokenizer.py

@@@FILE lite_llama/engine/llm_engine.py
"""Unified generation engine: one-shot batches over the executor.

:class:`LLMEngine` owns the tokenizer, model runner and executor end to

@@@FILE lite_llama/engine/prefix_cache.py
"""Prefix caching: reuse the KV of shared prompt prefixes across requests.

:class:`PrefixCache` hashes fixed-size blocks of tokens, matches a new
prompt's block run against cached ones, and reference-counts blocks so
several sequences share one physical KV region until all finish.

Usage:
    cache = PrefixCache(); match = cache.admit(token_ids)
    cache.release(token_ids)
"""

@@@FILE lite_llama/engine/sampler.py
"""Token sampling: turn a step's logits into the next token id per sequence.

:class:`Sampler` applies temperature, top-p nucleus sampling and repetition
penalty to a batch of logits; greedy decoding short-circuits to argmax. The
pure helpers below stay unit-testable on CPU without a model.

Usage:
    probs = torch.softmax(logits / temperature, -1)
    token = sample_top_p(probs, top_p, k)
"""
@@@END
@@@FILE lite_llama/engine/scheduler.py
"""Request scheduling with chunked prefill and prefix caching.

Per step the :class:`Scheduler` decides which requests prefill (in chunks),
which decode, and which slot each holds, so a long prompt cannot stall
running decodes. Scheduling commits: once planned, a chunk is assumed to
have run, and the request objects are the single source of truth.

Usage:
    scheduler = Scheduler(SchedulerConfig(...), num_slots)
    scheduler.add_request(request); plan = scheduler.schedule()
"""
@@@END
@@@FILE lite_llama/engine/stop_criteria.py
"""Device-resident stop bookkeeping for a decode batch (mirrors vLLM).

:class:`StopCriteria` holds one writable flag per sequence plus the batch's
stop-token rows, and after each step marks exactly the sequences whose new
token is a stop id. :func:`detect_repetition` is the CPU-side loop breaker.

Usage:
    criteria = StopCriteria(batch_size, stop_ids, vocab_size)
    detect_repetition(text, window, min_reps)
"""

@@@FILE lite_llama/executor/executor.py
"""Executor: the one interface an engine has for running a model pass.

:class:`Executor` is the abstract seam — ``execute`` a :class:`ModelInput`
in, sampled tokens out; :class:`UniProcExecutor` runs the worker in-process
while :class:`MultiprocExecutor` forwards plans to TP follower ranks.

Usage:
    executor = UniProcExecutor(engine, max_num_seqs, max_seq_len)
    tokens, logprobs = executor.execute(model_input)
"""

@@@FILE lite_llama/executor/overlap.py

@@@FILE lite_llama/executor/slot_batch.py

:class:`SlotBatch` gives every active sequence one KV row range ("slot")

Usage:
